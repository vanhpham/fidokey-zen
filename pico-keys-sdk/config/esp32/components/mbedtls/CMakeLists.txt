if(CONFIG_MBEDTLS_CERTIFICATE_BUNDLE)
    idf_build_get_property(python PYTHON)
    set(PICO_KEYS_MBEDTLS_BUNDLE_NAME "x509_crt_bundle")
    set(PICO_KEYS_MBEDTLS_BUNDLE_INPUTS ${PICO_KEYS_MBEDTLS_CRT_BUNDLE_DIR}/cacrt_all.pem ${PICO_KEYS_MBEDTLS_CRT_BUNDLE_DIR}/cacrt_local.pem)
    set(PICO_KEYS_MBEDTLS_BUNDLE_ARGS --input ${PICO_KEYS_MBEDTLS_BUNDLE_INPUTS} -q --max-certs "${CONFIG_MBEDTLS_CERTIFICATE_BUNDLE_MAX_CERTS}")
    set(PICO_KEYS_MBEDTLS_BUNDLE_GENERATOR ${python} ${PICO_KEYS_MBEDTLS_CRT_BUNDLE_DIR}/gen_crt_bundle.py)
    get_filename_component(PICO_KEYS_MBEDTLS_BUNDLE_FILE ${PICO_KEYS_MBEDTLS_BUNDLE_NAME} ABSOLUTE BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}")
    add_custom_command(OUTPUT ${PICO_KEYS_MBEDTLS_BUNDLE_FILE} COMMAND ${PICO_KEYS_MBEDTLS_BUNDLE_GENERATOR} ${PICO_KEYS_MBEDTLS_BUNDLE_ARGS} DEPENDS ${PICO_KEYS_MBEDTLS_BUNDLE_INPUTS} VERBATIM)
    add_custom_target(pico_keys_mbedtls_bundle DEPENDS ${PICO_KEYS_MBEDTLS_BUNDLE_FILE})
    add_dependencies(${COMPONENT_LIB} pico_keys_mbedtls_bundle)
    target_add_binary_data(${COMPONENT_LIB} ${PICO_KEYS_MBEDTLS_BUNDLE_FILE} BINARY)
    set_property(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" APPEND PROPERTY ADDITIONAL_CLEAN_FILES "${PICO_KEYS_MBEDTLS_BUNDLE_FILE}")
endif()
